# 宗门组织架构 — 技能树式 SVG 动态流光实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将宗门组织架构从纵向列表改造为从上到下的 SVG 动态流光树状图，掌门为树根，同级职务横向排列。

**Architecture:** 纯前端改造，只修改 index.html。用 CSS Flexbox 实现层级横向排列，SVG 绝对定位覆盖绘制连线，`stroke-dashoffset` 动画实现流光效果。掌门节点自动绑定玩家角色名。

**Tech Stack:** Vanilla HTML/CSS/JS，无新增依赖。

    function renderOrgStructureHTML(org) {
      const organization = org || [
        { rank: 0, name: '掌门', members: [] },
        { rank: 1, name: '副掌门', members: [] },
        { rank: 2, name: '长老', members: [] },
        { rank: 3, name: '真传弟子', members: [] },
        { rank: 4, name: '内门弟子', members: [] },
        { rank: 5, name: '外门弟子', members: [] },
        { rank: 6, name: '杂役', members: [] }
      ];
      const allMembers = window.gameStateManager?.state?.members || {};
      const playerName = window.gameStateManager?.state?.player?.name || '';

      // 按 rank 分组
      const rankGroups = {};
      organization.forEach(role => {
        if (!rankGroups[role.rank]) rankGroups[role.rank] = [];
        rankGroups[role.rank].push(role);
      });
      const sortedRanks = Object.keys(rankGroups).map(Number).sort((a, b) => a - b);

      // 确保 rank 0 存在
      if (!rankGroups[0]) {
        rankGroups[0] = [{ rank: 0, name: '掌门', members: [] }];
        sortedRanks.unshift(0);
      }

      let html = '<div class="org-tree" id="org-tree">';
      html += '<svg class="org-tree-svg" id="org-tree-svg"></svg>';
      html += '<div class="org-tree-levels">';

      sortedRanks.forEach(rank => {
        const roles = rankGroups[rank];
        html += `<div class="org-tree-level" data-rank="${rank}">`;

        roles.forEach(role => {
          const isLeader = role.rank === 0 && role.name === '掌门';
          const memberTags = (role.members || []).map(mn => {
            const member = allMembers[mn];
            const realm = member ? member.realm : '';
            return `<span class="org-node-member-tag" onclick="event.stopPropagation();showOrgMemberDetail('${mn}', '${role.name}')">${mn}${realm ? '（' + realm + '）' : ''}</span>`;
          }).join('');

          if (isLeader) {
            // 掌门节点：自动绑定玩家名，不可分配
            html += `
              <div class="org-tree-node org-tree-node--leader">
                <div class="org-node-badge">👑 掌门</div>
                <div class="org-node-title">${playerName || '未设定'}</div>
                <div class="org-node-members">
                  ${playerName ? '' : '<span class="org-node-empty">请在开局创建中设定掌门</span>'}
                </div>
              </div>
            `;
          } else {
            html += `
              <div class="org-tree-node" onclick="showOrgRoleAssign(${role.rank}, '${role.name}')">
                <div class="org-node-title">${role.name}</div>
                <div class="org-node-members">
                  ${memberTags || '<span class="org-node-empty">暂无</span>'}
                </div>
                <div class="org-node-actions">
                  ${role.name !== '掌门' ? `<button class="org-node-btn" onclick="event.stopPropagation();removeOrgRole(${role.rank}, '${role.name}')">删除</button>` : ''}
                </div>
              </div>
            `;
          }
        });

        // 当前 rank 添加同级职务按钮
        html += `<button class="org-node-btn" style="align-self:center;padding:8px 12px;border-radius:8px;" onclick="openAddOrgRoleModal(${rank})">+ 同级职务</button>`;
        html += '</div>';
      });

      html += '</div>'; // end org-tree-levels

      // 底部：新增层级按钮
      const maxRank = sortedRanks.length > 0 ? Math.max(...sortedRanks) : -1;
      html += `
        <div class="org-tree-add-btn">
          <button class="action-btn-sm" onclick="openAddOrgRoleModal(${maxRank + 1})">+ 新增层级（rank ${maxRank + 1}）</button>
        </div>
      `;

      html += '</div>'; // end org-tree
      return html;
    }
  ```

    function drawOrgTreeLines() {
      const tree = document.getElementById('org-tree');
      const svg = document.getElementById('org-tree-svg');
      if (!tree || !svg) return;

      const treeRect = tree.getBoundingClientRect();
      svg.setAttribute('width', tree.scrollWidth);
      svg.setAttribute('height', tree.scrollHeight);
      svg.style.width = tree.scrollWidth + 'px';
      svg.style.height = tree.scrollHeight + 'px';

      const levels = tree.querySelectorAll('.org-tree-level');
      if (levels.length < 2) return;

      // 定义渐变和滤镜
      let defs = svg.querySelector('defs');
      if (!defs) {
        defs = document.createElementNS('http://www.w3.org/2000/svg', 'defs');
        svg.appendChild(defs);
      }

      // 金色渐变（掌门 → 下一层）
      if (!document.getElementById('org-gold-gradient')) {
        const goldGrad = document.createElementNS('http://www.w3.org/2000/svg', 'linearGradient');
        goldGrad.id = 'org-gold-gradient';
        goldGrad.setAttribute('x1', '0%');
        goldGrad.setAttribute('y1', '0%');
        goldGrad.setAttribute('x2', '0%');
        goldGrad.setAttribute('y2', '100%');
        goldGrad.innerHTML = `
          <stop offset="0%" stop-color="rgba(200,168,110,0.6)"/>
          <stop offset="100%" stop-color="rgba(110,207,207,0.3)"/>
        `;
        defs.appendChild(goldGrad);
      }

      // 青色渐变（普通层级）
      if (!document.getElementById('org-jade-gradient')) {
        const jadeGrad = document.createElementNS('http://www.w3.org/2000/svg', 'linearGradient');
        jadeGrad.id = 'org-jade-gradient';
        jadeGrad.setAttribute('x1', '0%');
        jadeGrad.setAttribute('y1', '0%');
        jadeGrad.setAttribute('x2', '0%');
        jadeGrad.setAttribute('y2', '100%');
        jadeGrad.innerHTML = `
          <stop offset="0%" stop-color="rgba(110,207,207,0.5)"/>
          <stop offset="100%" stop-color="rgba(110,207,207,0.2)"/>
        `;
        defs.appendChild(jadeGrad);
      }

      // 发光滤镜
      if (!document.getElementById('org-glow')) {
        const filter = document.createElementNS('http://www.w3.org/2000/svg', 'filter');
        filter.id = 'org-glow';
        filter.innerHTML = `
          <feGaussianBlur stdDeviation="2" result="coloredBlur"/>
          <feMerge><feMergeNode in="coloredBlur"/><feMergeNode in="SourceGraphic"/></feMerge>
        `;
        defs.appendChild(filter);
      }

      // 清除旧连线
      const oldLines = svg.querySelectorAll('.org-tree-line');
      oldLines.forEach(l => l.remove());

      // 绘制连线：从上层每个节点中心 → 下层每个节点中心
      for (let i = 0; i < levels.length - 1; i++) {
        const parentLevel = levels[i];
        const childLevel = levels[i + 1];
        const parentNodes = parentLevel.querySelectorAll('.org-tree-node');
        const childNodes = childLevel.querySelectorAll('.org-tree-node');

        // 取父层和子层的中心参考点
        const parentRect = parentLevel.getBoundingClientRect();
        const childRect = childLevel.getBoundingClientRect();

        childNodes.forEach(childNode => {
          const childNodeRect = childNode.getBoundingClientRect();
          const childCx = childNodeRect.left + childNodeRect.width / 2 - treeRect.left;
          const childCy = childNodeRect.top - treeRect.top;

          // 找到最近的父节点
          let closestParent = null;
          let minDist = Infinity;
          parentNodes.forEach(pn => {
            const pnRect = pn.getBoundingClientRect();
            const pnCx = pnRect.left + pnRect.width / 2 - treeRect.left;
            const dist = Math.abs(pnCx - childCx);
            if (dist < minDist) {
              minDist = dist;
              closestParent = pn;
            }
          });

          if (closestParent) {
            const pRect = closestParent.getBoundingClientRect();
            const pCx = pRect.left + pRect.width / 2 - treeRect.left;
            const pCy = pRect.bottom - treeRect.top;

            // 二次贝塞尔曲线
            const midY = (pCy + childCy) / 2;
            const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
            const d = `M ${pCx} ${pCy} Q ${pCx} ${midY} ${childCx} ${childCy}`;
            path.setAttribute('d', d);
            path.setAttribute('fill', 'none');
            path.setAttribute('stroke-width', '1.5');
            path.setAttribute('stroke-linecap', 'round');
            path.setAttribute('filter', 'url(#org-glow)');
            path.setAttribute('class', 'org-tree-line');

            // 掌门连线用金色，其他用青色
            const isFromLeader = closestParent.classList.contains('org-tree-node--leader');
            path.setAttribute('stroke', isFromLeader ? 'url(#org-gold-gradient)' : 'url(#org-jade-gradient)');

            // 流光动画
            path.style.strokeDasharray = '8 4';
            path.style.animation = 'flowLight 3s linear infinite';

            svg.appendChild(path);
          }
        });
      }
    }
  ```

    function addOrgRole(name, targetRank) {
      const state = window.gameStateManager?.state;
      if (!state) return;
      const org = state.sect?.organization || [];
      const rank = typeof targetRank === 'number' ? targetRank : org.length;
      const newRole = { rank: rank, name: name, members: [] };
      window.gameStateManager.applyPatch([{
        op: 'insert',
        path: '/sect/organization/-',
        value: newRole
      }]);
      const content = `新增职位「${name}」（rank ${rank}）`;

      renderSectInfo();

### Task 5: 掌门节点数据同步

**Files:**
- Modify: `index.html`

- [ ] **Step 1: 确保掌门 members 自动同步玩家名**

  在 `renderSectInfo` 函数开头（获取 sect 和 player 之后），添加掌门数据自动同步：

  ```js
      // 同步掌门成员为玩家角色
      const org = sect.organization || [];
      const leaderRole = org.find(r => r.rank === 0 && r.name === '掌门');
      if (leaderRole && player.name) {
        const currentMembers = leaderRole.members || [];
        if (!currentMembers.includes(player.name)) {
          // 更新掌门成员
          const orgIdx = org.indexOf(leaderRole);
          window.gameStateManager.applyPatch([{
            op: 'replace',
            path: '/sect/organization/' + orgIdx + '/members',
            value: [player.name]
          }]);
        }
      }
  ```

  注意：这段代码需要在 `renderSectInfo` 渲染之前执行，但 `renderSectInfo` 本身会被频繁调用，需要避免无限循环。更好的方式是在获取 state 后检查是否需要同步，然后如果需要才 applyPatch。

  实际上，applyPatch 会触发 state 变更，可能导致重新渲染。为了避免问题，我们应该在渲染 HTML 之前先读取最新的 state，或者只在 state 初始化时同步一次。

  更安全的做法：在 `renderOrgStructureHTML` 中直接显示玩家名，而不修改 state。掌门节点的 members 数组为空也没关系，UI 上直接显示玩家名即可。

  所以这一步可以跳过——掌门节点已经在 `renderOrgStructureHTML` 中处理了。
