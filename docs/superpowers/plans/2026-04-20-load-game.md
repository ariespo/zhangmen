> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实装标题页「读取游戏」功能，按玩家名称分组，每个角色下有 N 个独立存档槽。

**Architecture:** 扩展 chat 对象的 `playerName` 字段作为分组依据；读档界面为左右分栏布局（左侧玩家列表 + 右侧存档卡片）；读档操作直接绑定到现有 IndexedDB + gameStateManager 体系。

**Tech Stack:** 纯前端 HTML/CSS/JS，Dexie.js (IndexedDB)，无测试框架。

### Task 2: 新增 game-state.js getAllSaves 函数

**Files:**
- Modify: `sillytavern/game-state.js`

- [ ] **Step 1: 在导出区新增 getAllSaves**

  在 `saveGameState` 之后、`applyPatches` 之前插入：

  ```js
  export async function getAllSaves() {
    try {
      const chats = await db.chats.toArray();
      const saves = chats.filter(c => c.playerName);
      const grouped = {};
      for (const chat of saves) {
        if (!grouped[chat.playerName]) grouped[chat.playerName] = [];
        // 提取展示信息
        const gs = chat.variables?.gameState || {};
        grouped[chat.playerName].push({
          id: chat.id,
          name: chat.name,
          playerName: chat.playerName,
          updatedAt: chat.updatedAt,
          sectName: gs.sect?.name || '未知宗门',
          playerRealm: gs.player?.realm || '未知',
          memberCount: Object.keys(gs.members || {}).length,
          messageCount: chat.messages?.length || 0
        });
      }
      for (const name in grouped) {
        grouped[name].sort((a, b) => b.updatedAt - a.updatedAt);
      }
      return grouped;
    } catch (e) {
      console.error('[GameState] getAllSaves failed:', e);
      return {};
    }
  }
  ```

### Task 4: 实现读档界面 JS 逻辑

**Files:**
- Modify: `index.html`（在 `window.loadGame` 函数及附近）

- [ ] **Step 1: 添加全局状态和方法**

  在 `window.loadGame` 附近（约第 6289 行）替换为以下代码：

  ```js
    let _currentLoadPlayer = null;
    let _loadGameData = {};

    window.loadGame = async function() {
      try {
        const { getAllSaves } = await import('./sillytavern/game-state.js');
        _loadGameData = await getAllSaves();
        const players = Object.keys(_loadGameData).sort();

        if (players.length === 0) {
          alert('暂无存档，请先开始新游戏');
          return;
        }

        _currentLoadPlayer = players[0];
        renderLoadGameScreen();
        document.getElementById('load-game-screen').style.display = 'flex';
      } catch (e) {
        console.error('[LoadGame] failed:', e);
        alert('读取存档列表失败');
      }
    };

    function hideLoadGameScreen() {
      document.getElementById('load-game-screen').style.display = 'none';
      _currentLoadPlayer = null;
      _loadGameData = {};
    }

    function renderLoadGameScreen() {
      const players = Object.keys(_loadGameData).sort();
      const playerList = document.getElementById('load-player-list');
      const saveList = document.getElementById('load-save-list');
      const saveHeading = document.getElementById('load-save-heading');
      const btnNewSave = document.getElementById('btn-new-save');

      // 渲染玩家列表
      playerList.innerHTML = players.map(name => {
        const count = _loadGameData[name].length;
        const activeClass = name === _currentLoadPlayer ? 'active' : '';
        return `<div class="load-player-item ${activeClass}" onclick="selectLoadPlayer('${name}')">
          <span>${name}</span>
          <span class="load-player-count">${count}</span>
        </div>`;
      }).join('');

      // 渲染存档列表
      const saves = _loadGameData[_currentLoadPlayer] || [];
      saveHeading.textContent = `${_currentLoadPlayer} 的存档`;
      btnNewSave.style.display = 'inline-block';

      if (saves.length === 0) {
        saveList.innerHTML = '<div class="load-empty">暂无存档</div>';
      } else {
        saveList.innerHTML = saves.map(save => {
          const date = new Date(save.updatedAt).toLocaleString('zh-CN', {
            month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit'
          });
          return `<div class="load-save-card">
            <div class="load-save-info">
              <div class="load-save-name">${save.name}</div>
              <div class="load-save-meta">
                <span>${save.sectName}</span>
                <span>${save.playerRealm}</span>
                <span>${save.memberCount} 弟子</span>
                <span>${date}</span>
              </div>
            </div>
            <div class="load-save-actions">
              <button class="btn-load" onclick="loadSave('${save.id}')">读取</button>
              <button class="btn-delete" onclick="deleteSave('${save.id}')">删除</button>
            </div>
          </div>`;
        }).join('');
      }
    }

    window.selectLoadPlayer = function(name) {
      _currentLoadPlayer = name;
      renderLoadGameScreen();
    };

    window.loadSave = async function(chatId) {
      try {
        // 加载游戏状态
        await window.gameStateManager.load(chatId);
        // 加载 SillyTavern 对话
        await window.sillyTavernStore.loadChat(chatId);
        // 刷新 UI
        refreshAllUI();
        // 进入游戏
        hideLoadGameScreen();
        document.getElementById('title-screen').classList.add('hidden');
        document.getElementById('app').style.display = 'flex';
        switchPage('story');
      } catch (e) {
        console.error('[LoadSave] failed:', e);
        alert('读取存档失败');
      }
    };

    window.deleteSave = async function(chatId) {
      if (!confirm('确定要删除这个存档吗？此操作不可撤销。')) return;
      try {
        await window.sillyTavernStore.deleteChat(chatId);
        // 如果删除的是当前活跃存档，重置 gameStateManager
        if (window.gameStateManager._chatId === chatId) {
          window.gameStateManager.reset();
        }
        // 刷新列表
        const { getAllSaves } = await import('./sillytavern/game-state.js');
        _loadGameData = await getAllSaves();
        const players = Object.keys(_loadGameData).sort();
        if (players.length === 0) {
          hideLoadGameScreen();
          alert('所有存档已删除');
          return;
        }
        if (!_loadGameData[_currentLoadPlayer]) {
          _currentLoadPlayer = players[0];
        }
        renderLoadGameScreen();
      } catch (e) {
        console.error('[DeleteSave] failed:', e);
        alert('删除存档失败');
      }
    };

    window.createNewSaveForCurrentPlayer = async function() {
      try {
        const playerName = _currentLoadPlayer;
        const saves = _loadGameData[playerName] || [];
        const saveName = `${playerName} - 存档${saves.length + 1}`;
        const chat = await window.sillyTavernStore.createChat(saveName, { playerName });
        // 初始化默认 gameState
        await window.gameStateManager.load(chat.id);
        // 刷新列表
        const { getAllSaves } = await import('./sillytavern/game-state.js');
        _loadGameData = await getAllSaves();
        renderLoadGameScreen();
      } catch (e) {
        console.error('[CreateSave] failed:', e);
        alert('新建存档失败');
      }
    };
  ```
